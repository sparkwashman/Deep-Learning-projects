from transformers import pipeline

# Load the text generation pipeline using the GPT-2 model
text_generator = pipeline('text-generation', model='gpt2')
def generate_text(prompt, max_length=50, temperature=0.7, top_k=50, top_p=0.9):
    """
    Generates text continuation based on a prompt and decoding parameters.
    """
    results = text_generator(
        prompt,
        max_new_tokens=max_length,
        num_return_sequences=1,
        temperature=temperature,
        top_k=top_k,
        top_p=top_p,
        do_sample=True,
        pad_token_id=50256 # Standard padding token for GPT-2
    )
    
    # Extract and return the generated text
    return results[0]['generated_text']
# Define a starting prompt
prompt = "In the future, artificial intelligence will"

print("--- Experiment 1: Standard Creative Generation ---")
print(generate_text(prompt, max_length=40, temperature=0.7))

print("\n--- Experiment 2: Focused / Deterministic Output (Low Temp) ---")
print(generate_text(prompt, max_length=40, temperature=0.2))

print("\n--- Experiment 3: High Creativity / Wild Output (High Temp) ---")
print(generate_text(prompt, max_length=40, temperature=1.2))
